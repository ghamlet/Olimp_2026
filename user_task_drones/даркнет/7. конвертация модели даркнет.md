### ONNX-модель — это два компонента:
### - Граф — описание слоёв и связей (какая свёртка где, вход/выход). Это тот самый 20-килобайтный yolov4_1_3_416_416_static.onnx.
### - Веса — сами обученные числа (все 24 МБ). В этом файле они лежат отдельно, во внешнем файле yolov4_1_3_416_416_static.onnx.data, на который граф ссылается.



```python

#!/usr/bin/env python3
import argparse
import time
import numpy as np
import cv2
import onnxruntime as ort

CAMERA_ID = "/dev/video0"
RESOLUTION = (1280, 960)
MODEL_PATH = "/home/pi/base/model/yolov4_1_3_416_416_static.onnx"
NAMES_PATH = "/home/pi/base/model/coco.names"

INPUT_SIZE = 416
CONF_THRESH = 0.4
NMS_THRESH = 0.45
PROCESS_EVERY = 3

ANCHORS = [12, 16, 19, 36, 40, 28, 36, 75, 76, 55, 72, 146, 142, 110, 192, 243, 459, 401]
GRID_LAYOUT = [(13, 6), (26, 3)]  # (grid, начальный индекс маски якорей)


def sigmoid(x):
    return 1.0 / (1.0 + np.exp(-x))


def decode_outputs(raw_boxes, raw_confs):
    """Декодирование сырых выходов ONNX (грид 13, stride 32 и грид 26, stride 16).

    Возвращает список [x1, y1, x2, y2, score, class_id] в относительных координатах (0..1).
    """
    dets = []

    for raw, grid, anc_start in ((raw_boxes, 13, 6), (raw_confs, 26, 3)):
        t = raw[0].reshape(3, 85, grid, grid)

        obj = sigmoid(t[:, 4])  # [3, grid, grid]
        mask = obj > CONF_THRESH
        anchors_idx, yy, xx = np.where(mask)

        for a, j, i in zip(anchors_idx, yy, xx):
            aw = ANCHORS[(anc_start + a) * 2]
            ah = ANCHORS[(anc_start + a) * 2 + 1]

            xs = sigmoid(t[a, 0, j, i])
            ys = sigmoid(t[a, 1, j, i])
            w = np.exp(t[a, 2, j, i]) * aw / INPUT_SIZE
            h = np.exp(t[a, 3, j, i]) * ah / INPUT_SIZE
            cx, cy = (i + xs) / grid, (j + ys) / grid

            cls = sigmoid(t[a, 5:, j, i])
            ci = int(np.argmax(cls))
            score = float(cls[ci] * obj[a, j, i])

            if score >= CONF_THRESH:
                dets.append((cx - w / 2, cy - h / 2, cx + w / 2, cy + h / 2, score, ci))

    if dets:
        boxes = np.array([[d[0], d[1], d[2] - d[0], d[3] - d[1]] for d in dets], dtype=np.float32)
        confs = np.array([d[4] for d in dets], dtype=np.float32)
        keep = cv2.dnn.NMSBoxes(boxes.tolist(), confs.tolist(), CONF_THRESH, NMS_THRESH)
        if len(keep) > 0:
            keep = np.array(keep).flatten()
            dets = [dets[int(i)] for i in keep]
    return dets


def parse_args():
    p = argparse.ArgumentParser(description="Детекция YOLOv4 (ONNX) с камеры, каждый 3-й кадр, с FPS")
    p.add_argument("--camera", default=CAMERA_ID, help=f"Устройство камеры (по умолч. {CAMERA_ID})")
    p.add_argument("--show", action="store_true", help="Показывать окно с кадра (нужен X/дисплей)")
    p.add_argument("--fps-window", type=float, default=1.0, help="Окно усреднения FPS, сек")
    return p.parse_args()


def main():
    args = parse_args()

    with open(NAMES_PATH, "r") as f:
        classes = [line.strip() for line in f.readlines()]

    session = ort.InferenceSession(MODEL_PATH)
    input_name = session.get_inputs()[0].name
    print(f"Модель: {MODEL_PATH} | вход: {session.get_inputs()[0].shape}\n")

    cap = cv2.VideoCapture(args.camera, cv2.CAP_V4L2)
    cap.set(cv2.CAP_PROP_FOURCC, cv2.VideoWriter_fourcc(*"MJPG"))
    cap.set(cv2.CAP_PROP_FRAME_WIDTH, RESOLUTION[0])
    cap.set(cv2.CAP_PROP_FRAME_HEIGHT, RESOLUTION[1])

    if not cap.isOpened():
        raise RuntimeError(f"Не удалось открыть камеру {args.camera}")

    frame_id = 0
    processed = 0
    start_time = time.time()
    fps_win_start = start_time
    fps_win_count = 0

    print("Старт распознавания (каждый 3-й кадр)... Ctrl+C для остановки\n")

    try:
        while cap.isOpened():
            ret, frame = cap.read()
            if not ret:
                print("Ошибка получения кадра со стрима.")
                break

            frame_id += 1
            now = time.time()
            dets = []

            if frame_id % PROCESS_EVERY == 0:
                processed += 1
                t0 = now

                rgb = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
                blob = cv2.resize(rgb, (INPUT_SIZE, INPUT_SIZE)).astype(np.float32) / 255.0
                blob = np.transpose(blob, (2, 0, 1))[None]

                outputs = session.run(None, {input_name: blob})
                dets = decode_outputs(outputs[0], outputs[1])

                infer_ms = (time.time() - t0) * 1000

                h, w = frame.shape[:2]
                for x1, y1, x2, y2, score, ci in dets:
                    X1 = int(x1 * w); Y1 = int(y1 * h)
                    X2 = int(x2 * w); Y2 = int(y2 * h)
                    cv2.rectangle(frame, (X1, Y1), (X2, Y2), (0, 255, 0), 2)
                    label = f"{classes[ci]} {score * 100:.0f}%"
                    cv2.putText(frame, label, (X1, max(20, Y1 - 8)),
                                cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 255, 0), 2)

                found = ", ".join(f"{classes[d[5]]}: {d[4] * 100:.1f}%" for d in dets) or "ничего"
                print(f"[Кадр {frame_id}] {found} | инференс {infer_ms:.0f} мс")

            fps_win_count += 1
            if now - fps_win_start >= args.fps_window:
                fps = fps_win_count / (now - fps_win_start)
                cv2.putText(frame, f"FPS: {fps:.1f}", (20, 40),
                            cv2.FONT_HERSHEY_SIMPLEX, 1.0, (0, 255, 0), 2)
                print(f"[FPS] {fps:.1f} (кадров за окно: {fps_win_count})")
                fps_win_start = now
                fps_win_count = 0

            if args.show:
                cv2.imshow("YOLOv4 ONNX", frame)
                if cv2.waitKey(1) & 0xFF == ord("q"):
                    break

    except KeyboardInterrupt:
        print("\nОстановка пользователем.")
    finally:
        cap.release()
        if args.show:
            cv2.destroyAllWindows()
        elapsed = time.time() - start_time
        print(f"\nИтого: кадров {frame_id}, обработано {processed}, "
              f"время {elapsed:.1f} с, общий FPS {frame_id / elapsed:.1f}")


if __name__ == "__main__":
    main()
```

### 2. Особенность моделей после `onnx2tf` (ВАЖНО)

Утилита `onnx2tf` при конвертации по умолчанию меняет порядок осей с **NCHW** `[1, 3, 416, 416]` (стандарт ONNX/PyTorch) на **NHWC** `[1, 416, 416, 3]` (стандарт TFLite).

  

Скрипт ниже автоматически определят формат входа вашей `.tflite` модели и сам изменит предобработку кадра.

  

### 3. Готовый скрипт `cam_tflite_fp32_detect.py`

```
#!/usr/bin/env python3
import argparse
import time
import cv2
import numpy as np

# Пробуем импортировать легкий tflite-runtime, либо полный tensorflow при наличии
try:
    import tflite_runtime.interpreter as tflite
except ImportError:
    import tensorflow.lite as tflite

CAMERA_ID = "/dev/video0"
RESOLUTION = (1280, 960)
MODEL_PATH = (
    "output_fp32/yolov4_1_3_416_416_static_float32.tflite"  # Путь к FP32 модели
)
NAMES_PATH = "/home/pi/base/model/coco.names"

CONF_THRESH = 0.4
NMS_THRESH = 0.45
PROCESS_EVERY = 3

ANCHORS = np.array(
    [
        12,
        16,
        19,
        36,
        40,
        28,
        36,
        75,
        76,
        55,
        72,
        146,
        142,
        110,
        192,
        243,
        459,
        401,
    ],
    dtype=np.float32,
)


def sigmoid(x):
    return 1.0 / (1.0 + np.exp(-x))


def decode_outputs_vectorized(raw_boxes, raw_confs, input_size):
    dets = []
    for raw, grid, anc_start in ((raw_boxes, 13, 6), (raw_confs, 26, 3)):
        # Приводим к виду [3, 85, grid, grid]
        t = raw.reshape(3, 85, grid, grid) if raw.ndim != 4 else raw[0]

        obj = sigmoid(t[:, 4])
        mask = obj > CONF_THRESH
        if not np.any(mask):
            continue

        anchors_idx, yy, xx = np.where(mask)
        obj_filtered = obj[anchors_idx, yy, xx]

        anc_pair_idx = (anc_start + anchors_idx) * 2
        aws = ANCHORS[anc_pair_idx]
        ahs = ANCHORS[anc_pair_idx + 1]

        xs = sigmoid(t[anchors_idx, 0, yy, xx])
        ys = sigmoid(t[anchors_idx, 1, yy, xx])
        w = np.exp(t[anchors_idx, 2, yy, xx]) * aws / input_size
        h = np.exp(t[anchors_idx, 3, yy, xx]) * ahs / input_size

        cx = (xx + xs) / grid
        cy = (yy + ys) / grid

        cls_logits = t[anchors_idx, 5:, yy, xx]
        cls_probs = sigmoid(cls_logits)
        cls_ids = np.argmax(cls_probs, axis=1)
        max_cls_probs = cls_probs[np.arange(len(cls_ids)), cls_ids]

        scores = max_cls_probs * obj_filtered

        valid = scores >= CONF_THRESH
        if np.any(valid):
            v_cx, v_cy = cx[valid], cy[valid]
            v_w, v_h = w[valid], h[valid]
            v_sc, v_ci = scores[valid], cls_ids[valid]

            for x, y, bw, bh, sc, ci in zip(
                v_cx, v_cy, v_w, v_h, v_sc, v_ci
            ):
                dets.append(
                    (
                        x - bw / 2,
                        y - bh / 2,
                        x + bw / 2,
                        y + bh / 2,
                        float(sc),
                        int(ci),
                    )
                )

    if dets:
        boxes = np.array(
            [[d[0], d[1], d[2] - d[0], d[3] - d[1]] for d in dets],
            dtype=np.float32,
        )
        confs = np.array([d[4] for d in dets], dtype=np.float32)
        keep = cv2.dnn.NMSBoxes(
            boxes.tolist(), confs.tolist(), CONF_THRESH, NMS_THRESH
        )
        if len(keep) > 0:
            keep = np.array(keep).flatten()
            dets = [dets[int(i)] for i in keep]
    return dets


def main():
    with open(NAMES_PATH, "r") as f:
        classes = [line.strip() for line in f.readlines()]

    # 1. Инициализация TFLite Интерпретатора
    interpreter = tflite.Interpreter(
        model_path=MODEL_PATH, num_threads=4  # Используем все 4 ядра Raspberry Pi
    )
    interpreter.allocate_tensors()

    input_details = interpreter.get_input_details()
    output_details = interpreter.get_output_details()

    input_shape = input_details[0]["shape"]
    print(f"Загружена модель TFLite FP32: {MODEL_PATH}")
    print(f"Формат входного тензора: {input_shape}")

    # Определяем формат расположения осей (NHWC или NCHW)
    is_nhwc = input_shape[-1] == 3
    input_size = input_shape[1] if is_nhwc else input_shape[2]

    cap = cv2.VideoCapture(CAMERA_ID, cv2.CAP_V4L2)
    cap.set(cv2.CAP_PROP_FOURCC, cv2.VideoWriter_fourcc(*"MJPG"))
    cap.set(cv2.CAP_PROP_FRAME_WIDTH, RESOLUTION[0])
    cap.set(cv2.CAP_PROP_FRAME_HEIGHT, RESOLUTION[1])

    if not cap.isOpened():
        raise RuntimeError(f"Не удалось открыть камеру {CAMERA_ID}")

    frame_id = 0
    last_dets = []
    print("Старт TFLite детекции... Нажмите Ctrl+C для выхода.")

    try:
        while cap.isOpened():
            ret, frame = cap.read()
            if not ret:
                break

            frame_id += 1
            if frame_id % PROCESS_EVERY == 0:
                t0 = time.time()

                # 2. Подготовка входных данных
                rgb = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
                resized = cv2.resize(rgb, (input_size, input_size))
                blob = resized.astype(np.float32) / 255.0

                if is_nhwc:
                    # Формат [1, 416, 416, 3]
                    input_data = np.expand_dims(blob, axis=0)
                else:
                    # Формат [1, 3, 416, 416]
                    input_data = np.transpose(blob, (2, 0, 1))[None]

                # 3. Инференс TFLite
                interpreter.set_tensor(input_details[0]["index"], input_data)
                interpreter.invoke()

                # 4. Получение результатов
                out_0 = interpreter.get_tensor(output_details[0]["index"])
                out_1 = interpreter.get_tensor(output_details[1]["index"])

                last_dets = decode_outputs_vectorized(out_0, out_1, input_size)

                infer_ms = (time.time() - t0) * 1000
                found = (
                    ", ".join(
                        f"{classes[d[5]]}: {d[4] * 100:.1f}%" for d in last_dets
                    )
                    or "ничего"
                )
                print(
                    f"[Кадр {frame_id}] {found} | TFLite FP32 инференс: {infer_ms:.0f} мс"
                )

            # Отрисовка
            h, w = frame.shape[:2]
            for x1, y1, x2, y2, score, ci in last_dets:
                X1, Y1 = int(x1 * w), int(y1 * h)
                X2, Y2 = int(x2 * w), int(y2 * h)
                cv2.rectangle(frame, (X1, Y1), (X2, Y2), (0, 255, 0), 2)
                cv2.putText(
                    frame,
                    f"{classes[ci]} {score * 100:.0f}%",
                    (X1, max(20, Y1 - 8)),
                    cv2.FONT_HERSHEY_SIMPLEX,
                    0.6,
                    (0, 255, 0),
                    2,
                )

    except KeyboardInterrupt:
        print("\nОстановлено пользователем.")
    finally:
        cap.release()


if __name__ == "__main__":
    main()
```

### Вариант 3. TFLite Full INT8 (Полное квантование весов и активаций)

Обеспечивает максимальную скорость на процессоре Raspberry Pi (задействует векторные инструкции ARM NEON). Требует небольшой выборки изображений (10–20 штук из датасета) для калибровки диапазона значений.

  

1. Создайте скрипт `convert_full_int8.py`:
    
      
    

Python

```
import glob
import cv2
import numpy as np
import tensorflow as tf

# Путь к сохраненной SavedModel после работы onnx2tf или напрямую через TFLiteConverter
ONNX_MODEL = "/path/to/yolov4_1_3_416_416_static.onnx"
IMAGES_DIR = (
    "/home/arrma/PROGRAMMS/Olimp_2026/DARKNET/user_task_drones/images/*.jpg"
)


# Генератор калибровочных данных
def representative_data_gen():
    image_paths = glob.glob(IMAGES_DIR)[:20]  # Берем первые 20 картинок
    for path in image_paths:
        img = cv2.imread(path)
        img = cv2.resize(img, (416, 416))
        img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
        img = img.astype(np.float32) / 255.0
        img = np.expand_dims(img, axis=0)
        yield [img]


# Сначала конвертируем ONNX в SavedModel через onnx2tf
# После этого запускаем квантование:
converter = tf.lite.TFLiteConverter.from_saved_model("saved_model")
converter.optimizations = [tf.lite.Optimize.DEFAULT]
converter.representative_dataset = representative_data_gen
converter.target_spec.supported_ops = [tf.lite.OpsSet.TFLITE_BUILTINS_INT8]
converter.inference_input_type = tf.float32  # или tf.int8
converter.inference_output_type = tf.float32

tflite_model = converter.convert()

with open("yolov4_tiny_full_int8.tflite", "wb") as f:
    f.write(tflite_model)

print("Полное INT8 квантование завершено: yolov4_tiny_full_int8.tflite")
```
